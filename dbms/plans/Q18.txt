-- Type: READ
-- Database: findback_db (PostgreSQL 16.15 (Ubuntu 16.15-0ubuntu0.24.04.1)); data: 400 reports, 30 users, 150 matches, 80 claims
-- Command: EXPLAIN (ANALYZE, BUFFERS) -- warm run

EXPLAIN (ANALYZE, BUFFERS)
SELECT
    cl."id" AS "claimId",
    cl."message",
    cl."status" AS "claimStatus",
    cl."createdAt",
    ir."id" AS "foundReportId",
    ir."title" AS "itemTitle",
    ir."imageUrl",
    l."name" AS "locationName",
    finder."name" AS "finderName"
FROM "Claim" cl
JOIN "ItemReport" ir ON ir."id" = cl."foundReportId"
JOIN "Location" l ON l."id" = ir."locationId"
JOIN "User" finder ON finder."id" = ir."userId"
WHERE cl."userId" = 'u3'
ORDER BY cl."createdAt" DESC;
