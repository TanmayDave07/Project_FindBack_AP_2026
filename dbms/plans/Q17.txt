-- Type: READ
-- Database: findback_db (PostgreSQL 16.15 (Ubuntu 16.15-0ubuntu0.24.04.1)); data: 400 reports, 30 users, 150 matches, 80 claims
-- Command: EXPLAIN (ANALYZE, BUFFERS) -- warm run

EXPLAIN (ANALYZE, BUFFERS)
SELECT
    cl."id" AS "claimId",
    cl."message",
    cl."status" AS "claimStatus",
    cl."createdAt",
    u."id" AS "claimantId",
    u."name" AS "claimantName",
    u."email" AS "claimantEmail",
    m."score" AS "matchScore",
    v."status" AS "verificationStatus"
FROM "Claim" cl
JOIN "User" u ON u."id" = cl."userId"
LEFT JOIN "Match" m ON m."id" = cl."matchId"
LEFT JOIN "Verification" v ON v."claimId" = cl."id"
WHERE cl."foundReportId" = 'r2'
ORDER BY cl."createdAt" ASC;
